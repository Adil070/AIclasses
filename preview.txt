        const canvas = document.getElementById('bg-canvas');
        const ctx = canvas.getContext('2d');
        let particles = [];

        function resizeCanvas() {
            canvas.width = window.innerWidth;
            canvas.height = window.innerHeight;
        }
        window.addEventListener('resize', resizeCanvas);
        resizeCanvas();

        class Particle {
            constructor() {
                this.x = Math.random() * canvas.width;
                this.y = Math.random() * canvas.height;
                this.size = Math.random() * 2 + 0.5;
                this.speedX = (Math.random() - 0.5) * 0.4;
                this.speedY = (Math.random() - 0.5) * 0.4;
                this.opacity = Math.random() * 0.5 + 0.2;
            }
            update() {
                this.x += this.speedX;
                this.y += this.speedY;

                if (this.x < 0) this.x = canvas.width;
                if (this.x > canvas.width) this.x = 0;
                if (this.y < 0) this.y = canvas.height;
                if (this.y > canvas.height) this.y = 0;
            }
            draw() {
                ctx.fillStyle = `rgba(99, 102, 241, ${this.opacity})`;
                ctx.beginPath();
                ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
                ctx.fill();
            }
        }

        for (let i = 0; i < 60; i++) {
            particles.push(new Particle());
        }

        function animateCanvas() {
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            particles.forEach(p => {
                p.update();
                p.draw();
            });
            
            // Draw faint ambient lines between close particles
            for(let a = 0; a < particles.length; a++) {
                for(let b = a; b < particles.length; b++) {
                    let dx = particles[a].x - particles[b].x;
                    let dy = particles[a].y - particles[b].y;
                    let dist = Math.sqrt(dx*dx + dy*dy);
                    if(dist < 100) {
                        ctx.strokeStyle = `rgba(99, 102, 241, ${0.08 - dist/1000})`;
                        ctx.lineWidth = 0.5;
                        ctx.beginPath();
                        ctx.moveTo(particles[a].x, particles[a].y);
                        ctx.lineTo(particles[b].x, particles[b].y);
                        ctx.stroke();
                    }
                }
            }
            requestAnimationFrame(animateCanvas);
        }
        animateCanvas();

        function openEnquireModal(preselectCourse = '') {
            if(preselectCourse) {
                document.getElementById('form-course').value = preselectCourse;
            }
            const modal = document.getElementById('enquire-modal');
            modal.classList.remove('hidden');
            setTimeout(() => modal.classList.remove('opacity-0'), 10);
        }

        function closeEnquireModal() {
            const modal = document.getElementById('enquire-modal');
            modal.classList.add('opacity-0');
            setTimeout(() => modal.classList.add('hidden'), 300);
        }

        function showToast(title, desc) {
            const toast = document.getElementById('toast-notification');
            document.getElementById('toast-title').innerText = title;
            document.getElementById('toast-desc').innerText = desc;

            toast.classList.remove('translate-y-20', 'opacity-0', 'pointer-events-none');
            setTimeout(() => {
                toast.classList.add('translate-y-20', 'opacity-0', 'pointer-events-none');
            }, 4000);
        }

        // Form Submission with Web3Forms / Formspree compatibility
        function handleInquirySubmit(e) {
            e.preventDefault();
            const name = document.getElementById('form-name').value;
            const phone = document.getElementById('form-phone').value;
            const course = document.getElementById('form-course').value;
            const timing = document.getElementById('form-timing').value;
            const message = document.getElementById('form-message').value;

            // Push into local inquiries array for Admin CMS
            studentInquiries.unshift({
                time: 'Just now',
                name: name,
                phone: phone,
                course: course,
                batch: timing,
                msg: message || 'No extra query provided.'
            });

            renderCmsInquiries();
            closeEnquireModal();
            document.getElementById('inquiry-form').reset();

            showToast('Enquiry Submitted Successfully!', 'Our Govandi team will call you shortly.');
        }

        function toggleViewMode() {
            const publicView = document.getElementById('public-view');
            const cmsView = document.getElementById('cms-view');
            const toggleText = document.getElementById('toggle-text');

            if (cmsView.classList.contains('hidden')) {
                publicView.classList.add('hidden');
                cmsView.classList.remove('hidden');
                toggleText.innerText = 'Switch to Public Site';
                renderCmsTable();
                renderCmsInquiries();
            } else {
                cmsView.classList.add('hidden');
                publicView.classList.remove('hidden');
                toggleText.innerText = 'Switch to Admin CMS';
            }
        }

        function renderCmsTable() {
            const tbody = document.getElementById('cms-courses-table');
            tbody.innerHTML = '';

            coursesData.forEach((course, index) => {
                const tr = document.createElement('tr');
                tr.className = 'hover:bg-white/5 transition-colors';
                tr.innerHTML = `
                    <td class="p-3 font-bold text-white">${course.title}</td>
                    <td class="p-3"><span class="px-2 py-0.5 rounded text-[10px] bg-indigo-500/20 text-indigo-300">${course.badge}</span></td>
                    <td class="p-3">${course.duration}</td>
                    <td class="p-3">${course.level}</td>
                    <td class="p-3">
                        <input type="number" value="${course.seats}" onchange="updateCourseSeats(${index}, this.value)" class="w-16 bg-slate-800 border border-white/20 rounded px-2 py-1 text-center font-bold text-emerald-400">
                    </td>
                    <td class="p-3">
                        <input type="text" value="${course.fee}" onchange="updateCourseFee(${index}, this.value)" class="w-20 bg-slate-800 border border-white/20 rounded px-2 py-1 text-center font-bold text-cyan-400">
                    </td>
                    <td class="p-3">
                        <button onclick="deleteCourse(${index})" class="text-rose-400 hover:text-rose-300 font-bold">
                            <i class="fa-solid fa-trash"></i>
                        </button>
                    </td>
                `;
                tbody.appendChild(tr);
            });
        }

        function updateCourseSeats(index, value) {
            coursesData[index].seats = parseInt(value) || 0;
            renderCourses();
            showToast('Seats Updated', `${coursesData[index].title} seats changed to ${value}`);
        }

        function updateCourseFee(index, value) {
            coursesData[index].fee = value;
            renderCourses();
            showToast('Fee Structure Updated', `${coursesData[index].title} fee set to ${value}`);
        }

        function deleteCourse(index) {
            if(confirm(`Are you sure you want to delete ${coursesData[index].title}?`)) {
                const name = coursesData[index].title;
                coursesData.splice(index, 1);
                renderCmsTable();
                renderCourses();
                showToast('Course Removed', `${name} deleted from catalogue.`);
            }
        }

        function createNewCoursePrompt() {
            const title = prompt('Enter Course Title (e.g. Artificial Intelligence Basics):');
            if(!title) return;
            const fee = prompt('Enter Course Fee (e.g. ₹10,000):', '₹8,500');

            coursesData.push({
                id: 'course-' + Date.now(),
                title: title,
                badge: 'NEW TRACK',
                category: 'Coding',
                duration: '3 Months',
                level: 'Beginner',
                seats: 5,
                fee: fee || '₹8,500',
                desc: 'Newly launched interactive certification course at Govandi, Mumbai branch.',
                icon: 'fa-solid fa-star',
                color: 'from-purple-500 to-indigo-500',
                salaryScope: '₹25,000 - ₹45,000 / month',
                modules: ['Core Concepts & Theory', 'Practical Hands-on Projects', 'Certification & Industry Prep']
            });

            renderCmsTable();
            renderCourses();
            showToast('New Course Added', `${title} is now active on the public website!`);
        }

        function renderCmsInquiries() {
            const tbody = document.getElementById('cms-inquiries-table');
            tbody.innerHTML = '';

            studentInquiries.forEach(inq => {
                const tr = document.createElement('tr');
                tr.innerHTML = `
                    <td class="p-3 text-gray-400">${inq.time}</td>
                    <td class="p-3 font-bold text-white">${inq.name}</td>
                    <td class="p-3 text-cyan-400 font-medium">${inq.phone}</td>
                    <td class="p-3"><span class="px-2 py-0.5 rounded text-[10px] bg-purple-500/20 text-purple-300 font-semibold">${inq.course}</span></td>
                    <td class="p-3 text-gray-300">${inq.batch}</td>
                    <td class="p-3 text-gray-400 italic">${inq.msg}</td>
                `;
                tbody.appendChild(tr);
            });
        }

        function updateCmsStats() {
            const students = document.getElementById('cms-stat-students').value;
            const courses = document.getElementById('cms-stat-courses').value;
            const exp = document.getElementById('cms-stat-exp').value;

            document.getElementById('stat-students').innerText = students;
            document.getElementById('stat-courses').innerText = courses;
            document.getElementById('stat-experience').innerText = exp;

            showToast('Stats Updated', 'Public banner key figures updated.');
        }

        // Initialize on Load
        window.onload = function() {
            renderCourses();
        };
